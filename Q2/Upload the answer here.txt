public class IT21802812Lab2Q2 {
    public static void main(String[] args) {
        double length = 10;
        double PI = 3.14;

        // The rope length equals the perimeter of the square
        double perimeter = 4 * length;

        // Same rope becomes the circumference of the circle
        // circumference = 2 * PI * radius
        double radius = perimeter / (2 * PI);

        System.out.println("Radius of the circular fence: " + radius);
    }
}
