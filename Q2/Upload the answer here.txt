public class IT23158290Lab2Q2 {
    public static void main(String[] args) {

        double length = 10.0;
        double PI = 3.14;

        double squarePerimeter = 4 * length;
        double radius = squarePerimeter / (2 * PI);

        System.out.println("Radius of the circular fence: " + radius);
    }
}
